# debian
```
site='https://sources.debian.org/data/main/m/mimetex/'
for ver in 1.50-1 1.73-2 1.74-1 1.76-1; do
  mkdir ${ver}
  cd "${ver}"
  wget "${site}/${ver}.checksums"
  for f in README gfuntype.c gifsave.c mimetex.c mimetex.h mimetex.html texfonts.h; do
    wget "${site}/${ver}/${f}"
  done
  cd ..
done
```
